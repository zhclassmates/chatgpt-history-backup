原因：你重装或重启了 sshd，主机指纹变了。SSH 在本机的 `known_hosts` 里还存着旧指纹，所以被拒。

按顺序做：

## 1）清掉旧指纹
mac 终端执行：
```bash
ssh-keygen -R 103.76.85.191
ssh-keygen -R "[103.76.85.191]:50455"   # 之前用过 50455 就一起删
# 可验证是否已删干净
ssh-keygen -F 103.76.85.191
```
如你只想直接删报错里的那一行（第 23 行）：
```bash
sed -i '' '23d' ~/.ssh/known_hosts   # macOS 的 sed 语法
```

## 2）校验新指纹（建议）
有控制台/面板可登录到服务器的话，在服务器上运行：
```bash
ssh-keygen -l -E sha256 -f /etc/ssh/ssh_host_ed25519_key.pub
```
输出应当与提示里看到的：
```
SHA256:y2hyAsOim4mQNGmPQrjyizpz94+rtoGtFBLUW3PchPE
```
一致再信任。如果没法上控制台，可临时用本机获取并查看（注意这步不是绝对防 MITM）：
```bash
ssh-keyscan -t ed25519 103.76.85.191 | ssh-keygen -lf - -E sha256
```

## 3）加入新指纹并连接
确认无误后连接并自动接受新指纹：
```bash
ssh -o StrictHostKeyChecking=accept-new \
    -o IPQoS=none -o ServerAliveInterval=15 -o ServerAliveCountMax=3 \
    root@103.76.85.191 -p 22
```
若你已改回自定义端口，再把 `-p 22` 换成 `-p 50455`。

也可先预写入：
```bash
ssh-keyscan -t ed25519 103.76.85.191 >> ~/.ssh/known_hosts
ssh root@103.76.85.191
```

## 4）进机后再做的稳固项
仅在已登录后执行，避免锁死自己。
```bash
# 确认允许 root 与公钥登录
grep -E '^(PermitRootLogin|PubkeyAuthentication|PasswordAuthentication|Port)' /etc/ssh/sshd_config || true

# 如要固定端口 50455：
sed -i 's/^#\?Port .*/Port 50455/' /etc/ssh/sshd_config
ufw allow 50455/tcp
systemctl reload sshd
```

- 你前面的 “Connection closed by remote host” 出现在 `apt` 拉包后，通常是 `openssh-server` 重启导致会话被切断，属正常。
- 指纹变更后，先清 `known_hosts` 再连是标准处理。未经校验直接接受新指纹有中间人风险，最好用控制台对比指纹。

def okx_get_order(apiKey, secretKey, passphrase, instId, ordId):
    if not (instId and ordId and passphrase):
        raise HTTPException(400, "OKX 需要 instId、ordId、passphrase")
    ts = datetime.utcnow().isoformat(timespec="milliseconds") + "Z"
    method = "GET"
    path = f"/api/v5/trade/order"
    q = f"instId={instId}&ordId={ordId}"
    prehash = f"{ts}{method}{path}?{q}"
    sign = base64.b64encode(hmac.new(secretKey.encode(), prehash.encode(), hashlib.sha256).digest()).decode()
    headers = {
        "OK-ACCESS-KEY": apiKey,
        "OK-ACCESS-SIGN": sign,
        "OK-ACCESS-TIMESTAMP": ts,
        "OK-ACCESS-PASSPHRASE": passphrase,
        "Content-Type": "application/json",

def binance_get_order(apiKey, secretKey, symbol, orderId, isFutures=True):
    if not (symbol and orderId):
        raise HTTPException(400, "Binance 需要 symbol、orderId")
    base = "https://fapi.binance.com" if isFutures else "https://api.binance.com"
    path = "/fapi/v1/order" if isFutures else "/api/v3/order"

    params = f"symbol={symbol}&orderId={orderId}&timestamp={ts}"
    sig = hmac.new(secretKey.encode(), params.encode(), hashlib.sha256).hexdigest()
    url = f"{base}{path}?{params}&signature={sig}"
    headers = {"X-MBX-APIKEY": apiKey}
    r = requests.get(url, headers=headers, timeout=15)
    if not r.ok:
        raise HTTPException(r.status_code, f"Binance错误: {r.text}")
    return r.json()

    return {
        "exchange": "okx",
        "ordId": o.get("ordId"),
        "instId": o.get("instId"),
        "side": o.get("side"),
        "state": o.get("state"),
        "accFillSz": o.get("accFillSz"),
        "avgPx": o.get("avgPx"),
        "px": o.get("px"),
        "sz": o.get("sz"),
        "tdMode": o.get("tdMode"),
        "uTime": o.get("uTime"),
    }

def normalize_binance(raw: Dict[str, Any]) -> Dict[str, Any]:
    return {
        "exchange": "binance",
        "orderId": raw.get("orderId"),
        "symbol": raw.get("symbol"),
        "side": raw.get("side"),
        "status": raw.get("status"),
        "origQty": raw.get("origQty"),
        "executedQty": raw.get("executedQty"),
        "avgPrice": raw.get("avgPrice") or raw.get("price"),
        "updateTime": raw.get("updateTime") or raw.get("time"),

    norm_str = json.dumps(normalized, sort_keys=True, separators=(",",":"))
    root = "0x" + keccak(text=norm_str).hex()

    inst = ids.get("instId") or ids.get("symbol") or ""
    leaf = f"{exchange}:{inst}:{oid}"
    orderIdHash = "0x" + keccak(text=leaf).hex()
    return {
        "version":"v1",
        "orderIdHash": orderIdHash,
        "root": root,
        "leaves": [leaf],
        "normalized_json": normalized,
        "raw_snapshot": rawAny,
        "ts": now_ts(),
    }

@app.get("/healthz")

    return {"status":"ok","ts":now_ts(),"service":"jp-verify","version":"v1"}

@app.post("/api/verify")

        norm = normalize_okx(raw)

    else:

        ev = make_evidence("binance", {"orderId":str(req.orderId),"symbol":req.symbol}, norm, raw)
