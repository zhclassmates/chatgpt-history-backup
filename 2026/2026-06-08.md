## 4. 注意：朗读的是“句中形式”，不是原形

这个很重要。

比如：

```text
Guten → 朗读 Guten
原形显示 gut

geht → 朗读 geht
原形显示 gehen

Ihnen → 朗读 Ihnen
原形显示 Sie / Ihnen
```

不能点 `Guten` 却读 `gut`。  
因为你是在学句子里的真实发音。

## 第三步：生成单词 MP3 脚本

新建文件：

```text

```

放进去：

```python
import asyncio
import json
import re
from pathlib import Path
import edge_tts

BASE_DIR = Path("/Users/zhaomosheng/Desktop/123")
WORDS_FILE = BASE_DIR / "words.txt"
OUT_DIR = BASE_DIR / "word_audio"
MANIFEST_FILE = BASE_DIR / "word_manifest.json"

VOICE = "de-DE-KatjaNeural"
RATE = "-10%"

OUT_DIR.mkdir(parents=True, exist_ok=True)


def clean_word(w: str) -> str:
    w = w.strip()
    w = re.sub(r"^[\"'„“”‚‘’\(\)\[\]\{\},.!?:;]+", "", w)
    w = re.sub(r"[\"'„“”‚‘’\(\)\[\]\{\},.!?:;]+$", "", w)
    return w.strip()


def load_words():
    raw = WORDS_FILE.read_text(encoding="utf-8").splitlines()
    words = []
    seen = set()

    for line in raw:
        word = clean_word(line)
        if not word:
            continue

        key = word.lower()
        if key not in seen:
            seen.add(key)
            words.append(word)

    return words


async def generate_one(word: str, filename: str):
    out_path = OUT_DIR / filename

    if out_path.exists() and out_path.stat().st_size > 0:
        print(f"跳过已有：{filename} {word}")
        return

    text = word
    communicate = edge_tts.Communicate(
        text=text,
        voice=VOICE,
        rate=RATE
    )

    await communicate.save(str(out_path))
    print(f"生成：{filename} {word}")


async def main():
    words = load_words()
    manifest = []

    print(f"总单词数：{len(words)}")

    for i, word in enumerate(words, start=1):
        filename = f"word_{i:04d}.mp3"

        try:
            await generate_one(word, filename)
            manifest.append({
                "id": i,
                "text": word,
                "audio": filename
            })

            await asyncio.sleep(0.4)

        except Exception as e:
            print(f"失败：{word} -> {e}")

    MANIFEST_FILE.write_text(
        json.dumps(manifest, ensure_ascii=False, indent=2),
        encoding="utf-8"
    )

    print("完成")
    print(f"音频目录：{OUT_DIR}")
    print(f"清单文件：{MANIFEST_FILE}")


if __name__ == "__main__":
    asyncio.run(main())
```

```swift
final class SentenceAudioPlayer: NSObject, AVAudioPlayerDelegate {
    var player: AVAudioPlayer?
    var onFinish: (() -> Void)?

    func play(id: Int, slow: Bool) throws {
        stop()

        let session = AVAudioSession.sharedInstance()
        try session.setCategory(.playback, mode: .default)
        try session.setActive(true)

        guard let url = Bundle.main.url(
            forResource: "\(id)",
            withExtension: "mp3",
            subdirectory: "audio"
        ) else {
            throw NSError(
                domain: "AudioMissing",
                code: 404,
                userInfo: [NSLocalizedDescriptionKey: "找不到 audio/\(id).mp3"]
            )
        }

        player = try AVAudioPlayer(contentsOf: url)
        player?.delegate = self
        player?.enableRate = true
        player?.rate = slow ? 0.82 : 1.0
        player?.prepareToPlay()
        player?.play()
    }

    func stop() {
        if player?.isPlaying == true {
            player?.stop()
        }
        player = nil
    }

    func audioPlayerDidFinishPlaying(_ player: AVAudioPlayer, successfully flag: Bool) {
        DispatchQueue.main.async {
            self.onFinish?()
        }
    }
}
```
