# 📄 GPT-3.5 批量文本处理脚本 + Markdown 输出增强
# 功能：
# 1. 批量读取文本文件（每个文件一个文本）
# 2. 使用 GPT-3.5 进行：标点修复 + 错别字校正 + 分段 + 加标题
# 3. 自动生成 Markdown 文件，包含封面元信息（标题、作者、日期等）

import os
import openai
import datetime

# ✅ 1. 设置你的 API Key（建议使用环境变量存储）
openai.api_key = "sk-你的API密钥"

# ✅ 2. 输入/输出目录设置
INPUT_DIR = "/Users/shengmo/Desktop/mp3_txts"
OUTPUT_DIR = "/Users/shengmo/Desktop/output_texts"
os.makedirs(OUTPUT_DIR, exist_ok=True)

# ✅ 3. 定义 GPT 处理函数
def process_text(text, title="生成内容", author="GPT批处理助手"):
    prompt = (
        "请对以下中文文本进行增强处理，包括：\n"
        "1. 添加合适的标点符号\n"
        "2. 修正错别字与语法问题\n"
        "3. 分段，并为每段添加小标题（用【标题】格式）\n"
        "请直接返回优化后的文本：\n"
        f"{text}"
    )

    try:
        response = openai.ChatCompletion.create(
            model="gpt-3.5-turbo",
            messages=[{"role": "user", "content": prompt}],
            temperature=0.7,
            max_tokens=2048
        )
        output = response.choices[0].message.content

        # ✅ 转换【标题】为 Markdown 二级标题
        lines = output.splitlines()
        markdown_lines = []
        for line in lines:
            if line.startswith("【") and line.endswith("】"):
                h2 = line.strip("【】")
                markdown_lines.append(f"## {h2}")
            else:
                markdown_lines.append(line)

        # ✅ 添加 Markdown 元信息（YAML front matter）
        now = datetime.datetime.now().strftime("%Y-%m-%d %H:%M:%S")
        front_matter = [
            "---",
            f"title: {title}",
            f"author: {author}",
            f"date: {now}",
            "tags: [文本增强, GPT处理]",
            "lang: zh",
            "---",
            ""
        ]

        return "\n".join(front_matter + markdown_lines)
    except Exception as e:
        return f"【处理失败】: {str(e)}"

# ✅ 4. 批量读取 input 目录下所有 .txt 文件
for filename in os.listdir(INPUT_DIR):
    if filename.endswith(".txt"):
        with open(os.path.join(INPUT_DIR, filename), "r", encoding="utf-8") as f:
            raw_text = f.read()

        print(f"\n🚀 正在处理: {filename}...")
        title_base = os.path.splitext(filename)[0]
        output = process_text(raw_text, title=title_base)

        md_filename = title_base + ".md"
        with open(os.path.join(OUTPUT_DIR, md_filename), "w", encoding="utf-8") as f:
            f.write(output)
        print(f"✅ 输出完成: {md_filename}\n")

print("🎉 所有文本处理完毕！")
