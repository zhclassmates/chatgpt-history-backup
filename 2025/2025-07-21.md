## 📦 四、JS 中获取授权并处理逻辑

```js
Page({
  data: {},

  onLoad() {
    this.checkCameraPermission();
  },

  checkCameraPermission() {
    wx.getSetting({
      success: (res) => {
        if (!res.authSetting['scope.camera']) {
          // 未授权，主动请求
          wx.authorize({
            scope: 'scope.camera',
            success: () => {
              console.log('摄像头授权成功');
            },
            fail: () => {
              wx.showModal({
                title: '授权提示',
                content: '需要开启摄像头权限才能进行面试录制',
                success(res) {
                  if (res.confirm) {
                    wx.openSetting(); // 跳转授权设置
                  }
                }
              });
            }
          });
        }
      }
    });
  },

  onCameraError(e) {
    console.error('摄像头错误：', e.detail);
    wx.showToast({
      title: '摄像头异常',
      icon: 'none'
    });
  }
});
```

## ❗ 2. `net::ERR_NAME_NOT_RESOLVED` 和 `net::ERR_CONNECTION_REFUSED`

### 🔍 问题说明：
所有请求都无法连接服务器，报错 `ERR_NAME_NOT_RESOLVED` 和 `ERR_CONNECTION_REFUSED`。

### 📉 原因可能包括：

- 服务器地址配置错误或域名无法解析；
- 当前网络（如公司 WiFi、防火墙）限制了访问；
- 小程序没有配置合法域名（微信后台设置）；
- 请求走了 `http://localhost`、`127.0.0.1`、或不合法路径；

### ✅ 修复建议：

1. **检查请求地址**，确保你访问的 API 是公网域名，格式类似：

```js
https://api.example.com/path
```

2. **确认微信后台合法域名已添加：**

进入 [微信公众平台](https://mp.weixin.qq.com/) → 设置 → 服务器域名 → 添加：

- request 合法域名：你的 API 地址；
- download/upload/websocket 也需按需配置；

3. **开发阶段使用本地调试：**
- 使用本地代理或 `ngrok` 将本地服务映射到公网；
- 避免直接使用 `localhost` 或 `127.0.0.1`；

### ✅ 2. `ERR_NAME_NOT_RESOLVED` 与 `ERR_CONNECTION_REFUSED`
**原因：**
- API 请求的 URL 无法解析或连接；

**解决方案：**
- 在 `api.js` 中确认请求地址是否为公网地址：

// routes/speech.js
const express = require('express');
const router = express.Router();
const tencentcloud = require("tencentcloud-sdk-nodejs");

const TtsClient = tencentcloud.tts.v20190823.Client;
const clientConfig = {
  credential: {
    secretId: process.env.TENCENT_SECRET_ID,
    secretKey: process.env.TENCENT_SECRET_KEY,
  },
  region: "ap-guangzhou",
  profile: {
    httpProfile: {
      endpoint: "tts.tencentcloudapi.com",
    },
  },
};
const client = new TtsClient(clientConfig);

router.post('/speak', async (req, res) => {
  const { text } = req.body;
  const params = {
    Text: text,
    SessionId: "test-session",
    ModelType: 1,
    VoiceType: 1002,
    Codec: "mp3",
    Speed: 0,
    Volume: 5,
    PrimaryLanguage: 1,
    SampleRate: 16000,
  };

  try {
    const data = await client.TextToVoice(params);
    res.json({
      audioUrl: `data:audio/mp3;base64,${data.Audio}`,
    });
  } catch (err) {
    console.error(err);
    res.status(500).json({ error: "语音合成失败" });
  }
});

module.exports = router;
