### 示例代码（Python, 用 PyMuPDF）

```python
import fitz  # PyMuPDF
import os

input_folder = "/path/to/pdf_folder"
output_folder = "/path/to/output_folder"
os.makedirs(output_folder, exist_ok=True)

# 定义区域（要根据实际情况调整坐标）
big_title_area = (200, 500, 800, 550)  # 大图标题栏位置
small_clip_area = (100, 700, 600, 820) # 小图裁剪区域
write_position = (110, 710)            # 小图第一行文字位置

for file in os.listdir(input_folder):
    if not file.endswith(".pdf"):
        continue

    pdf_path = os.path.join(input_folder, file)
    doc = fitz.open(pdf_path)

    new_doc = fitz.open()  # 新PDF文档

    for page in doc:
        # ① 提取大图文字
        title_text = page.get_textbox(big_title_area).strip()

        # ② 裁剪小图
        rect = fitz.Rect(small_clip_area)
        pix = page.get_pixmap(clip=rect)
        img_pdf = fitz.open()  
        img_page = img_pdf.new_page(width=rect.width, height=rect.height)
        img_page.insert_image(img_page.rect, pixmap=pix)

        # ③ 在小图上写文字
        if title_text:
            img_page.insert_text(write_position, title_text, fontsize=10, fontname="helv")

        # 添加到新文档
        new_doc.insert_pdf(img_pdf)
        img_pdf.close()

    # 保存替换后的PDF
    output_path = os.path.join(output_folder, file)
    new_doc.save(output_path)
    new_doc.close()
    doc.close()

print("所有PDF已处理完成！")
```

```python
import os
```

```python
CHANNEL_URL = "https://www.youtube.com/@dlw2023/streams"   # 频道直播回放

    "outtmpl": os.path.join(DOWNLOAD_DIR, "%(upload_date>%Y-%m-%d)s - %(title)s.%(ext)s"),  # 文件命名：日期+标题
    "format": "mp4/best",      # 下载最佳 mp4 格式
    "ignoreerrors": True,      # 忽略错误继续下载

    "retries": 10,
    "fragment_retries": 10,
    "noprogress": False,
    "continuedl": True,

def download_channel():
    with yt_dlp.YoutubeDL(ydl_opts) as ydl:
        ydl.download([CHANNEL_URL])

if __name__ == "__main__":
    download_channel()
```

def format_timestamp(seconds):
    """将秒数格式化为SRT时间戳格式 HH:MM:SS,mmm"""
    td = datetime.timedelta(seconds=seconds)
    milliseconds = round((seconds - int(seconds)) * 1000)
    
    hours, remainder = divmod(td.seconds, 3600)
    minutes, seconds = divmod(remainder, 60)
    
    return f"{hours:02d}:{minutes:02d}:{seconds:02d},{milliseconds:03d}"

def write_srt(segments, file, to_simplified=True):
    """将segments写入SRT格式的文件"""
    for i, segment in enumerate(segments, 1):
        start = format_timestamp(segment["start"])
        end = format_timestamp(segment["end"])
        text = segment["text"].strip()
        
        # 如果需要，将文本转换为简体中文
        if to_simplified:
            text = cc.convert(text)
        
        file.write(f"{i}\n")
        file.write(f"{start} --> {end}\n")
        file.write(f"{text}\n\n")

```python
#!/usr/bin/env python
# -*- coding: utf-8 -*-

import os
import whisper
import subprocess
import tempfile
from opencc import OpenCC
import datetime

# 繁简转换器
cc = OpenCC('t2s')

def format_timestamp(seconds):
    td = datetime.timedelta(seconds=seconds)
    milliseconds = round((seconds - int(seconds)) * 1000)
    hours, remainder = divmod(td.seconds, 3600)
    minutes, seconds = divmod(remainder, 60)
    return f"{hours:02d}:{minutes:02d}:{seconds:02d},{milliseconds:03d}"

def write_srt(segments, file, to_simplified=True):
    for i, seg in enumerate(segments, 1):
        start = format_timestamp(seg["start"])
        end = format_timestamp(seg["end"])
        text = seg["text"].strip()
        if to_simplified:
            text = cc.convert(text)
        file.write(f"{i}\n{start} --> {end}\n{text}\n\n")

def extract_subtitle(video_path, model_name="medium", to_simplified=True):
    if not os.path.exists(video_path):
        print(f"文件不存在: {video_path}")
        return
    
    base_name = os.path.splitext(video_path)[0]
    output_path = f"{base_name}.srt"

    print(f"正在处理: {video_path}")
    model = whisper.load_model(model_name)

    # 提取音频
    audio_path = tempfile.mktemp(suffix=".wav")
    subprocess.run([
        "ffmpeg", "-i", video_path, "-vn", "-acodec", "pcm_s16le",
        "-ar", "16000", "-ac", "1", audio_path
    ], check=True)

    # 转写
    result = model.transcribe(audio_path, language="zh")
    
    # 写入字幕
    with open(output_path, "w", encoding="utf-8") as f:
        write_srt(result["segments"], f, to_simplified)
    
    os.unlink(audio_path)
    print(f"生成字幕: {output_path}")

def process_folder(folder_path, model_name="medium", to_simplified=True):
    for root, dirs, files in os.walk(folder_path):
        for file in files:
            if file.lower().endswith((".mp4", ".mkv", ".mov", ".avi")):
                video_path = os.path.join(root, file)
                try:
                    extract_subtitle(video_path, model_name, to_simplified)
                except Exception as e:
                    print(f"处理失败 {video_path}: {e}")

if __name__ == "__main__":
    import argparse
    parser = argparse.ArgumentParser(description="批量视频转字幕（支持递归文件夹）")
    parser.add_argument("folder", help="包含视频文件的文件夹路径")
    parser.add_argument("-m", "--model", default="medium", choices=["tiny","base","small","medium","large"], help="Whisper模型大小")
    parser.add_argument("--traditional", action="store_true", help="保留繁体中文")
    args = parser.parse_args()

    process_folder(args.folder, model_name=args.model, to_simplified=not args.traditional)
```

```python
    base_name = os.path.splitext(video_path)[0]

    # 检查是否已经生成对应格式字幕
    already_done = True
    for fmt in formats:
        if not os.path.exists(f"{base_name}.{fmt}"):
            already_done = False
```

确保这个路径是真实存在且有视频文件（`.mp4`、`.mkv` 等）。如果路径不存在，脚本会直接结束，因为 `os.walk()` 没有找到文件。

## 2️⃣ 没有打印信息
在我的脚本里，`extract_subtitle` 会打印：

```python
print(f"正在处理: {video_path}")
```

如果你什么都没看到，有可能是**视频文件不在支持的扩展名中**，或者文件夹里都是子文件夹而没有直接的视频文件。  
脚本默认只处理：

```python
(".mp4", ".mkv", ".mov", ".avi")
```

✅ 解决方法：  
在 `process_folder` 中加入调试打印：

```python
print(f"检查文件夹: {root}")
print(f"找到文件: {files}")
```

这样可以确认脚本是否进入了文件夹。

## 3️⃣ ffmpeg 是否可用
脚本需要 ffmpeg 提取音频，如果 ffmpeg 不可用，可能会抛异常，但你没看到，因为 `stdout` 和 `stderr` 被重定向了：

```python
stdout=subprocess.DEVNULL, stderr=subprocess.DEVNULL
```

临时解决方法：  
把 `DEVNULL` 去掉，能看到错误信息：

```python
subprocess.run([
    "ffmpeg", "-i", video_path, "-vn", "-acodec", "pcm_s16le",
    "-ar", "16000", "-ac", "1", audio_path
], check=True)
```

然后重新执行，看看是否 ffmpeg 提示错误。

## 4️⃣ 视频太大，转写慢
使用 `-m large` 模型时，如果视频很长，Whisper 加载和转写会非常耗时（几十秒到几分钟不等）。  
在终端可能没有提示，你可以在脚本里加入更多打印：

```python
print("加载模型中...")
model = whisper.load_model(model_name)
print("模型加载完成，开始转写音频...")
```

这样就能确认是“模型加载中”还是“真的没进入循环”。
